% print-ready-header.tex
% ─────────────────────────────────────────────────────────────────────────────
% LaTeX header injected via pandoc --include-in-header for CompassCrew PDF exports.
%
% SHIPPED WITH THE PLUGIN (2026-08-22). Before this, pdf-export.ts fell back to
% ~/gitrepos/faerie2/forensics/publication-renders/print-ready-header.tex -- a path in a repo
% that is two rebrands gone. fs.existsSync() returned false, the --include-in-header argument
% was silently dropped, and every export lost float pinning, needspace discipline, link
% coloring and caption styling while reporting complete success. The "beautiful PDF" claim was
% resting on a file nobody had.
%
% Vendored from reckon: forensics/publication-renders/print-ready-header.tex.
%
% Provides:
%   1. No-split floats (figures + tables always placed [H])
%   2. Needspace discipline for headings that must not orphan
%   3. Hyperref with warm-professional colorlinks
%   4. Bibliography colorlinks matching body link color
%   5. Long-table support for genuinely large tables
%   6. Tight caption styling above tables, below figures
%   7. Code block frame + background for readability
% ─────────────────────────────────────────────────────────────────────────────

% ── Float placement ──────────────────────────────────────────────────────────
\usepackage{float}
\floatplacement{figure}{H}
\floatplacement{table}{H}

% ── Needspace: prevent orphaned headings ─────────────────────────────────────
\usepackage{needspace}
% Ensure at least 4 lines follow a section heading before a page break
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{1.5ex plus 0.5ex minus 0.2ex}{0.8ex plus 0.2ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus 0.4ex minus 0.2ex}{0.6ex plus 0.2ex}
\titlespacing*{\subsubsection}{0pt}{1.0ex plus 0.3ex minus 0.1ex}{0.4ex plus 0.1ex}

% Patch sectioning commands to require needspace
\let\oldsection\section
\renewcommand{\section}[1]{\needspace{4\baselineskip}\oldsection{#1}}
\let\oldsubsection\subsection
\renewcommand{\subsection}[1]{\needspace{3\baselineskip}\oldsubsection{#1}}

% ── Hyperref — warm professional colorlinks ───────────────────────────────────
% hyperref is loaded by pandoc; we override its options here
\PassOptionsToPackage{
  colorlinks=true,
  linkcolor=NavyBlue,
  urlcolor=NavyBlue,
  citecolor=NavyBlue,
  filecolor=NavyBlue,
  bookmarks=true,
  bookmarksnumbered=true,
  bookmarksopen=true,
  pdfborder={0 0 0},
  pdfdisplaydoctitle=true
}{hyperref}

% ── Caption styling ───────────────────────────────────────────────────────────
\usepackage[
  tableposition=top,
  figureposition=bottom,
  font=small,
  labelfont=bf,
  labelsep=period
]{caption}

% ── Long table support ────────────────────────────────────────────────────────
% longtable is already loaded by pandoc for markdown tables;
% we add booktabs for professional horizontal rules
\usepackage{booktabs}
\usepackage{longtable}
\setlength{\LTpre}{6pt}
\setlength{\LTpost}{6pt}

% ── Code block styling (verbatim + shaded) ────────────────────────────────────
\usepackage{fancyvrb}
\usepackage{framed}

% Light warm-grey background for code blocks (tango highlight style from pandoc)
\definecolor{shadecolor}{RGB}{248,248,244}
\renewenvironment{Shaded}{\begin{snugshade}}{\end{snugshade}}

% Ensure code blocks don't split across pages (use [H] via float + minipage)
% pandoc wraps Shaded in a minipage when --variable block-headings is set;
% for extra safety we add page-break-before=avoid equivalent:
\usepackage{etoolbox}
\AtBeginEnvironment{Shaded}{\needspace{3\baselineskip}}

% ── Page geometry refinements ─────────────────────────────────────────────────
% Widows and orphans — prevent single-line runts at top/bottom of page
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000

% Paragraph spacing — slightly looser for academic readability
\setlength{\parskip}{4pt plus 1pt minus 1pt}
\setlength{\parindent}{0pt}

% ── Section numbering depth ───────────────────────────────────────────────────
\setcounter{tocdepth}{3}
\setcounter{secnumdepth}{3}

% ── Footnote / endnote clickability ───────────────────────────────────────────
% hyperref handles footnote links automatically when colorlinks=true;
% we just ensure footnote marks are styled consistently
\usepackage[hang,flushmargin]{footmisc}
\setlength{\footnotemargin}{1.5em}

% ── PDF metadata ─────────────────────────────────────────────────────────────
% Filled in dynamically by the export script via --metadata flags;
% this block ensures the fields are accepted by hyperref
\AtBeginDocument{%
  \hypersetup{%
    pdfauthor={Amanda Morton; Claude Opus 4.7; the Swarmy collective},%
    pdfcreator={pandoc + xelatex + swarmy pdf-exporter},%
    pdfproducer={swarmy publication pipeline 2026-05-25}%
  }%
}
